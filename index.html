Docker and deployment

Containerised stack: FastAPI agent service, Qdrant for vectors, and an Nginx front end.

docker-compose.yml

services:
  api:
    build: ./api            # FastAPI + LangGraph agents
    environment: [LLM_API_KEY, QDRANT_URL=http://qdrant:6333]
    depends_on: [qdrant]
  qdrant:
    image: qdrant/qdrant:latest
    volumes: ["qdata:/qdrant/storage"]
  web:
    image: nginx:alpine
    volumes: ["./index.html:/usr/share/nginx/html/index.html:ro"]
    ports: ["8080:80"]
volumes: {qdata: {}}

Dockerfile (api)

FROM python:3.12-slim
WORKDIR /app
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt
COPY . .
CMD ["uvicorn","main:app","--host","0.0.0.0","--port","8000"]
